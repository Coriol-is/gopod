package ipc_test

import (
	"context"
	"encoding/json"
	"errors"
	"io"
	"log/slog"
	"os"
	"path/filepath"
	"sync/atomic"
	"testing"
	"time"

	"github.com/Coriol-is/gopod/internal/ipc"
	"github.com/Coriol-is/gopod/internal/scheduler"
	"github.com/Coriol-is/gopod/internal/store"
)

// Temporary diagnostic: assert the reported losing interleaving actually exists.
func TestDiagnosticReplayReplacementArchivedWithoutReceipt(t *testing.T) {
	root := t.TempDir()
	log := slog.New(slog.NewTextHandler(io.Discard, nil))
	st, err := store.Open(t.Context(), filepath.Join(root, "store.sqlite"), log)
	if err != nil {
		t.Fatal(err)
	}
	defer st.Close()
	if err = st.RegisterChat(t.Context(), store.RegisteredChat{JID: "tg:2", Folder: "alice", AddedAt: 1}); err != nil {
		t.Fatal(err)
	}
	base := filepath.Join(root, "ipc", "alice")
	if err = os.MkdirAll(filepath.Join(base, "tasks"), 0700); err != nil {
		t.Fatal(err)
	}
	input := filepath.Join(base, "tasks", "replay.json")
	receipt := filepath.Join(base, "results", "replay.result.json")
	paused, release := make(chan struct{}), make(chan struct{})
	var held atomic.Bool
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	w := ipc.New(ipc.Config{DataDir: root, OwnerFolder: "owner", Tick: 5 * time.Millisecond, NamespacePath: func(folder string) (string, bool) {
		if _, err := os.Stat(receipt); err == nil && held.CompareAndSwap(false, true) {
			close(paused)
			<-release
		}
		return filepath.Join(root, "ipc", folder), true
	}}, noMessages{}, scheduler.NewIPCSink(st), log, func(folder string) (string, bool) { return "tg:2", folder == "alice" })
	done := make(chan struct{})
	go func() { defer close(done); w.Run(ctx) }()
	defer func() { cancel(); <-done }()
	raw, _ := json.Marshal(schedule("daily"))
	if err = os.WriteFile(input, raw, 0600); err != nil {
		t.Fatal(err)
	}
	select {
	case <-paused:
	case <-time.After(3 * time.Second):
		close(release)
		t.Fatal("not paused before archive")
	}
	if err = os.Remove(receipt); err != nil {
		close(release)
		t.Fatal(err)
	}
	if err = os.WriteFile(input+".tmp", raw, 0600); err != nil {
		close(release)
		t.Fatal(err)
	}
	if err = os.Rename(input+".tmp", input); err != nil {
		close(release)
		t.Fatal(err)
	}
	close(release)
	deadline := time.Now().Add(time.Second)
	for time.Now().Before(deadline) {
		if _, err = os.Stat(input); errors.Is(err, os.ErrNotExist) {
			break
		}
		time.Sleep(time.Millisecond)
	}
	time.Sleep(30 * time.Millisecond)
	if _, err = os.Stat(input); !errors.Is(err, os.ErrNotExist) {
		t.Fatal("replacement not consumed", err)
	}
	if _, err = os.Stat(receipt); !errors.Is(err, os.ErrNotExist) {
		t.Fatal("race not reproduced: receipt exists", err)
	}
	tasks, err := st.ListTasksByChat(t.Context(), "alice")
	if err != nil || len(tasks) != 1 {
		t.Fatalf("mutation duplicated: %d %v", len(tasks), err)
	}
	t.Log("confirmed: replacement archived, receipt absent, exactly one task")
}
